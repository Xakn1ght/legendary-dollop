        t4.SetAnim("Head", opt#1)
# ==== GROUP: Create a centipede of a given size ====
WHEN System.IsGroupActive("Create a centipede of a given size"):
WHEN t5.OnFunction("CreateCentipede"):
  VAR uid = 0
  VAR y = 0
  VAR x = 0
  WHEN System.EveryTick():
      System.SetVar(gvar:x, (floor(random(1, ($HCELLS - 1))) * $CELLSIZE))
      System.SetVar(gvar:y, ($TOPMARGIN + $CELLSIZE))
  WHEN System.Compare(t5.Param(0), op>, 1):
    WHEN System.For("", 1, (t5.Param(0) - 1)):
        System.CreateObject(OBJ:t4, layer:1, $x, $y)
        t4.SetInstanceVar(ivar#2, $uid)
        t4.SetBoolInstanceVar(ivar#0, opt#0)
        System.SetVar(gvar:uid, t4.UID())
        t4.SetAnim("Body", opt#1)
        t4.SetInstanceVar(ivar#3, (($fastSpeed != 0) ? 400 : 300))
  WHEN System.EveryTick():
      System.CreateObject(OBJ:t4, layer:1, $x, $y)
      t4.SetInstanceVar(ivar#2, $uid)
      t4.SetAnim("Head", opt#1)
      t4.SetInstanceVar(ivar#4, choose(-1, 1))
      t4.SetInstanceVar(ivar#3, (($fastSpeed != 0) ? 400 : 300))
WHEN t4.OnCreated():
    t4.SetInstanceVar(ivar#6, t4.X())
    t4.SetInstanceVar(ivar#7, t4.Y())
# ==== GROUP: Segment Moving Code ====
WHEN System.IsGroupActive("Segment Moving Code"):
WHEN System.CompareVar(gvar:isPlaying, op!=, 0)  AND  System.ForEachOrdered(OBJ:t4, (t4.var0 ? 1 : 2), opt#0):
  VAR isDownMove = 0
  VAR pixDistance = 0
  WHEN t4.CompareX(op=, t4.var6)  AND  t4.CompareY(op=, t4.var7)  AND  t4.IsBoolInstanceVarSet(ivar#0):
      System.SetVar(gvar:isDownMove, 0)
      t5.CallFunction("PassNewTargetDown", args(t4.var2, t4.X(), t4.Y()))
      t4.MoveToTop()
    WHEN t4.IsOverlappingOffset(OBJ:t0, 0, (t4.var4 * 4))  OR  t4.IsBoolInstanceVarSet(ivar#1):
        System.SetVar(gvar:isDownMove, (-t4.var4))
    WHEN t4.IsOverlapping(OBJ:t4):
        System.SetVar(gvar:isDownMove, choose(-1, 1))
    WHEN t4.CompareX(op=, (($HCELLS - 1) * $CELLSIZE))  AND  t4.CompareInstanceVar(ivar#4, op>, 0):
        System.SetVar(gvar:isDownMove, -1)
    WHEN t4.CompareX(op=, $CELLSIZE)  AND  t4.CompareInstanceVar(ivar#4, op<, 0):
        System.SetVar(gvar:isDownMove, 1)
    WHEN System.CompareVar(gvar:isDownMove, op=, 0):
        t4.AddInstanceVar(ivar#6, ($CELLSIZE * t4.var4))
    WHEN System.Else():
        t4.AddInstanceVar(ivar#7, ($CELLSIZE * t4.var5))
        t4.SetInstanceVar(ivar#4, $isDownMove)
      WHEN t4.CompareInstanceVar(ivar#7, op=, ($TOPMARGIN + (($PLAYERAREA + $PLAYFIELD) * $CELLSIZE))):
          t4.SetInstanceVar(ivar#5, -1)
          t4.SetBoolInstanceVar(ivar#1, opt#0)
      WHEN t4.CompareInstanceVar(ivar#7, op=, ($TOPMARGIN + ($PLAYFIELD * $CELLSIZE))):
          t4.SetInstanceVar(ivar#5, 1)
  WHEN System.EveryTick():
      System.SetVar(gvar:pixDistance, (dt() * t4.var3))
      t4.SetX(((min(abs((t4.X() - t4.var6)), $pixDistance) * ((t4.X() < t4.var6) ? 1 : -1)) + t4.X()))
      t4.SetY(((min(abs((t4.Y() - t4.var7)), $pixDistance) * ((t4.Y() < t4.var7) ? 1 : -1)) + t4.Y()))
WHEN System.Else():
WHEN t5.OnFunction("PassNewTargetDown"):
  WHEN t4.PickByUID(t5.Param(0)):
      t4.SetInstanceVar(ivar#6, t5.Param(1))
      t4.SetInstanceVar(ivar#7, t5.Param(2))
      t5.CallFunction("PassNewTargetDown", args(t4.var2, t4.X(), t4.Y()))
# ==== GROUP: Function to create all Centipedes for a level ====
WHEN System.IsGroupActive("Function to create all Centipedes for a level"):
WHEN t5.OnFunction("CreateLevel"):
    System.SetVar(gvar:fastSpeed, 0)
    System.CreateObject(OBJ:t52, layer:"Objects", (layoutwidth() / 2), (layoutheight() / 2))
    t52.SetText(("LEVEL " & $level))

  WHEN t5.CompareParam(0, op>, 1)  AND  System.Compare((t5.Param(0) % 2), op=, 1):
      System.SetVar(gvar:fastSpeed, 1)
  WHEN t5.CompareParam(0, op>, 1):
    WHEN System.For("", 1, floor((t5.Param(0) / 2))):
        t5.CallFunction("CreateCentipede", args(1))
  WHEN t5.CompareParam(0, op<, 18):
      t5.CallFunction("CreateCentipede", args((9 - floor((t5.Param(0) / 2)))))

# SHEET: State
##########################################################################################
VAR state = 0
VAR STATE_NEWLEVEL = 1
VAR STATE_PLAY = 2
VAR STATE_LIFELOST = 3
VAR STATE_GAMEOVER = 4
# ==== GROUP: Initialisation and Tick ====
WHEN System.IsGroupActive("Initialisation and Tick"):
WHEN System.OnLayoutStart():
    System.SetVar(gvar:state, $STATE_NEWLEVEL)
WHEN System.EveryTick():
    System.SetVar(gvar:isPlaying, ($state == $STATE_PLAY))
# ==== GROUP: New Level State ====
WHEN System.IsGroupActive("New Level State"):
WHEN System.CompareVar(gvar:state, op=, $STATE_NEWLEVEL):
    t5.CallFunction("CreateLevel", args(((($level - 1) % 19) + 1)))
    System.SetVar(gvar:state, $STATE_PLAY)
# ==== GROUP: Play State ====
WHEN System.IsGroupActive("Play State"):
WHEN System.CompareVar(gvar:state, op=, $STATE_PLAY):
  WHEN t1.OnCollision(OBJ:t4)  OR  t1.OnCollision(OBJ:t67):
      System.SetVar(gvar:state, $STATE_LIFELOST)
      t1.Timer.StopTimer("beat")
WHEN System.CompareVar(gvar:state, op=, $STATE_PLAY):
  WHEN System.Compare(t4.Count(), op=, 0):
      System.AddVar(gvar:level, 1)
      System.SetVar(gvar:state, $STATE_NEWLEVEL)
      t1.Timer.StopTimer("beat")
WHEN System.TriggerOnce()  AND  System.CompareVar(gvar:state, op=, $STATE_PLAY):
    t1.Timer.StartTimer(0.4, opt#1, "beat")
WHEN t1.Timer.OnTimer("beat"):
    t20.Play(["centipede_beat", false], opt#0, 0, "")
# ==== GROUP: Life Lost State ====
WHEN System.IsGroupActive("Life Lost State"):
WHEN System.CompareVar(gvar:state, op=, $STATE_LIFELOST)  AND  System.TriggerOnce():
    System.SubVar(gvar:lives, 1)
    t5.CallFunction("RedrawLives", args())
    t67.Destroy()
    t31.Destroy()
    t1.Spawn(OBJ:t14, layer:1, 0)
    t1.SetVisible(opt#0)
    t20.Play(["centipede_death", false], opt#0, 0, "")
  WHEN System.PickAll(OBJ:t4):
      t4.Destroy()
  VAR fireTime = 0
  WHEN System.ForEachOrdered(OBJ:t0, (t0.X() + (t0.Y() * 9999)), opt#0):
      t0.Spawn(OBJ:t10, layer:0, 0)
      System.AddVar(gvar:fireTime, 0.03)
      t10.Timer.StartTimer($fireTime, opt#0, "start")
      System.CreateObject(OBJ:t65, layer:"Objects", (layoutwidth() / 2), (layoutheight() / 2))
      System.CreateObject(OBJ:t66, layer:"Objects", (layoutwidth() / 2), ((layoutheight() / 2) + 60))
      t66.SetText($score)
WHEN t10.Timer.OnTimer("start"):
    t10.SetVisible(opt#1)
    t10.Fade.StartFade()
    t5.CallFunction("AddToScore", args(2))
    t20.Play(["matching_combo_1", false], opt#0, 0, "")
    t66.SetText($score)
WHEN System.CompareVar(gvar:state, op=, $STATE_LIFELOST)  AND  System.Compare(t10.Count(), op=, 0):
    t53.Destroy()
    t1.SetVisible(opt#1)
  WHEN System.CompareVar(gvar:lives, op>, 0):
      System.SetVar(gvar:state, $STATE_NEWLEVEL)
  WHEN System.Else():
      System.SetVar(gvar:state, $STATE_GAMEOVER)
# ==== GROUP: Game Over State ====
WHEN System.IsGroupActive("Game Over State"):
WHEN System.CompareVar(gvar:state, op=, $STATE_GAMEOVER)  AND  System.TriggerOnce():
    System.SetLayerVisible(layer:"GameOver", opt#1)
    t35.SetText($score)
    t36.SetText($HighScore)
    t44.arcadeActionGameOver("{}", "{}")
WHEN t44.arcadeEventStart():
    System.SetLayerVisible(layer:"GameOver", opt#0)
    System.RestartLayout()
